class Cake {
  taste() {
    var adjective = "delicious";
    print "The " + this.flavor + " cake is " + adjective + "!";
  }
}

class T {
  taste() {
    var adjective = " super delicious";
    print "The " + this.flavor + " cake is " + adjective + "!";
  }
}


var cake = Cake();
var cheesCake = T();

cake.flavor = "German chocolate";
cheesCake.flavor = "Cheese";

cake.taste(); 
cheesCake.taste();
